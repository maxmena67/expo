import React, { useState } from 'react';
import { StyleSheet, View, Text, TouchableOpacity } from 'react-native';

export default function VotoGlobalAppMobile() {
  const [hasVoted, setHasVoted] = useState(false);
  const [votes, setVotes] = useState({ option1: 0, option2: 0, option3: 0 });

  const handleVote = (option) => {
    if (hasVoted) return;
    setVotes({ ...votes, [option]: votes[option] + 1 });
    setHasVoted(true);
  };

  const totalVotes = votes.option1 + votes.option2 + votes.option3;
  const getPercentage = (count) =>
    totalVotes === 0 ? 0 : ((count / totalVotes) * 100).toFixed(1);

  return (
    <View style={styles.container}>
      <Text style={styles.title}>VotoGlobal</Text>
      <Text style={styles.description}>
        ¿Quién debería liderar el mundo en 2050?
      </Text>

      {!hasVoted ? (
        <View style={styles.buttonContainer}>
          <TouchableOpacity
            style={[styles.button, styles.blueButton]}
            onPress={() => handleVote('option1')}
          >
            <Text style={styles.buttonText}>Una inteligencia artificial ética</Text>
          </TouchableOpacity>
          <TouchableOpacity
            style={[styles.button, styles.greenButton]}
            onPress={() => handleVote('option2')}
          >
            <Text style={styles.buttonText}>Un consejo ciudadano global</Text>
          </TouchableOpacity>
          <TouchableOpacity
            style={[styles.button, styles.purpleButton]}
            onPress={() => handleVote('option3')}
          >
            <Text style={styles.buttonText}>Un líder humano con valores planetarios</Text>
          </TouchableOpacity>
        </View>
      ) : (
        <View style={styles.resultsContainer}>
          <Text style={styles.resultsTitle}>Resultados Globales:</Text>
          <View style={styles.resultsBarContainer}>
            <ResultBar
              label="IA ética"
              value={votes.option1}
              percentage={getPercentage(votes.option1)}
              color="#60a5fa" // Equivalente a bg-blue-600
            />
            <ResultBar
              label="Consejo ciudadano"
              value={votes.option2}
              percentage={getPercentage(votes.option2)}
              color="#34d399" // Equivalente a bg-green-600
            />
            <ResultBar
              label="Líder humano"
              value={votes.option3}
              percentage={getPercentage(votes.option3)}
              color="#8b5cf6" // Equivalente a bg-purple-600
            />
          </View>
        </View>
      )}
    </View>
  );
}

function ResultBar({ label, percentage, value, color }) {
  return (
    <View style={styles.resultItem}>
      <View style={styles.resultLabel}>
        <Text style={styles.resultLabelText}>{label}</Text>
        <Text style={styles.resultPercentage}>{percentage}% ({value})</Text>
      </View>
      <View style={styles.progressBarBackground}>
        <View style={[styles.progressBar, { backgroundColor: color, width: ${percentage}% }]} />
      </View>
    </View>
  );
}

const styles = StyleSheet.create({
  container: {
    flex: 1,
    alignItems: 'center',
    justifyContent: 'center',
    padding: 20,
    backgroundColor: '#e0f2f7', // Un fondo claro similar
  },
  title: {
    fontSize: 24,
    fontWeight: 'bold',
    marginBottom: 20,
    textAlign: 'center',
  },
  description: {
    textAlign: 'center',
    fontSize: 16,
    marginBottom: 30,
    maxWidth: 300,
  },
  buttonContainer: {
    width: '100%',
    maxWidth: 300,
    gap: 15,
  },
  button: {
    padding: 15,
    borderRadius: 12,
    alignItems: 'center',
    shadowColor: '#000',
    shadowOffset: { width: 0, height: 2 },
    shadowOpacity: 0.23,
    shadowRadius: 2.62,
    elevation: 4,
  },
  blueButton: { backgroundColor: '#60a5fa' },
  greenButton: { backgroundColor: '#34d399' },
  purpleButton: { backgroundColor: '#8b5cf6' },
  buttonText: {
    color: 'white',
    fontWeight: 'bold',
    fontSize: 16,
  },
  resultsContainer: {
    marginTop: 30,
    width: '100%',
    maxWidth: 300,
  },
  resultsTitle: {
    fontSize: 18,
    fontWeight: 'semibold',
    textAlign: 'center',
    marginBottom: 15,
  },
  resultsBarContainer: {
    gap: 10,
  },
  resultItem: {
    gap: 5,
  },
  resultLabel: {
    flexDirection: 'row',
    justifyContent: 'space-between',
    fontSize: 14,
    marginBottom: 3,
  },
  resultLabelText: {
    fontWeight: 'bold',
  },
  resultPercentage: {
    color: 'gray',
  },
  progressBarBackground: {
    backgroundColor: '#e5e7eb', // Equivalente a bg-gray-200
    height: 6,
    borderRadius: 3,
    overflow: 'hidden',
  },
  progressBar: {
    height: '100%',
    borderRadius: 3,
  },
});
